Create a base protocol and screen objects for semantic element access:

```swift
import XCTest

/// Base protocol for screen objects providing semantic element access
protocol AppScreen {
    var app: XCUIApplication { get }
}

extension AppScreen {
    /// Find element by accessibility identifier with wait
    func element(_ identifier: String, timeout: TimeInterval = 5) -> XCUIElement {
        let element = app.descendants(matching: .any)[identifier]
        _ = element.waitForExistence(timeout: timeout)
        return element
    }

    /// Check if element exists (no wait)
    func exists(_ identifier: String) -> Bool {
        app.descendants(matching: .any)[identifier].exists
    }
}

/// Home screen elements
struct HomeScreen: AppScreen {
    let app: XCUIApplication

    var createHighlightButton: XCUIElement { element("createHighlightButton") }
    var settingsButton: XCUIElement { element("settingsButton") }

    func projectCard(id: String) -> XCUIElement {
        element("projectCard_\(id)")
    }
}
```

Create utility functions for common test operations:

```swift
import XCTest

extension XCTestCase {
    /// Wait for element to appear and be hittable
    func waitForHittable(_ element: XCUIElement, timeout: TimeInterval = 10) -> Bool {
        let predicate = NSPredicate(format: "exists == true AND isHittable == true")
        let expectation = XCTNSPredicateExpectation(predicate: predicate, object: element)
        let result = XCTWaiter.wait(for: [expectation], timeout: timeout)
        return result == .completed
    }

    /// Wait for element to disappear
    func waitForDisappear(_ element: XCUIElement, timeout: TimeInterval = 10) -> Bool {
        let predicate = NSPredicate(format: "exists == false")
        let expectation = XCTNSPredicateExpectation(predicate: predicate, object: element)
        let result = XCTWaiter.wait(for: [expectation], timeout: timeout)
        return result == .completed
    }

    /// Wait for any element with matching identifier prefix
    func waitForElementWithPrefix(_ app: XCUIApplication, prefix: String, timeout: TimeInterval = 10) -> XCUIElement? {
        let predicate = NSPredicate(format: "identifier BEGINSWITH %@", prefix)
        let elements = app.descendants(matching: .any).matching(predicate)
        let firstElement = elements.firstMatch
        if firstElement.waitForExistence(timeout: timeout) {
            return firstElement
        }
        return nil
    }
}
```
